-- ══════════════════════════════════════════
-- کۆدی ١ – CREATE DATABASE
-- ══════════════════════════════════════════
-- دروستکردنی بنکەی زانیاری نوێ
CREATE DATABASE database_name;

-- ══════════════════════════════════════════
-- کۆدی ٢ – DROP DATABASE
-- ══════════════════════════════════════════
-- سڕینەوەی بنکەی زانیاری بە تەواوی
DROP DATABASE database_name;

-- ══════════════════════════════════════════
-- کۆدی ٣ – SELECT
-- ══════════════════════════════════════════
-- هەموو ستوونەکان هەڵبژێرە
SELECT * FROM table_name;
-- ستوونی دیاریکراو هەڵبژێرە
SELECT column_names FROM table_name;

-- ══════════════════════════════════════════
-- کۆدی ٤ – WHERE
-- ══════════════════════════════════════════
-- تەنها ئەو ڕیزانە پیشان بدە کە مەرجەکە تێدایەتەوە
SELECT * FROM table_name
WHERE condition;

-- ══════════════════════════════════════════
-- کۆدی ٥ – UPDATE
-- ══════════════════════════════════════════
-- گۆڕینی ستوونێک یان چەند ستوون لە خشتەیەکدا
UPDATE table_name
SET column1 = value1, column2 = value2
WHERE condition;

-- ══════════════════════════════════════════
-- کۆدی ٦ – DELETE
-- ══════════════════════════════════════════
-- سڕینەوەی ڕیزی دیاریکراو بە مەرج
DELETE FROM table_name
WHERE condition;

-- ══════════════════════════════════════════
-- کۆدی ٧ – INSERT INTO
-- ══════════════════════════════════════════
-- زیادکردنی ڕیزی نوێ بۆ خشتەیەک
INSERT INTO table_name (column1, column2, column3)
VALUES (value1, value2, value3);

-- ══════════════════════════════════════════
-- کۆدی ٨ – CREATE TABLE & ALTER TABLE
-- ══════════════════════════════════════════
-- دروستکردنی خشتەی نوێ
CREATE TABLE table_name (
  column1 datatype constraint,
  column2 datatype constraint
);
-- زیادکردنی ستوونی نوێ
ALTER TABLE table_name ADD column_name datatype;
-- سڕینەوەی ستوون
ALTER TABLE table_name DROP COLUMN column_name;
-- گۆڕینی ناوی ستوون
ALTER TABLE table_name RENAME COLUMN old_name TO new_name;

-- ══════════════════════════════════════════
-- کۆدی ٩ – MODIFY, ADD CONSTRAINT & RENAME TABLE
-- ══════════════════════════════════════════
-- گۆڕینی جۆری داتا یان سنووری ستوون
ALTER TABLE table_name MODIFY column_name new_datatype constraint;
-- زیادکردنی سنووری نوێ
ALTER TABLE table_name ADD CONSTRAINT constraint_name constraint_definition;
-- گۆڕینی ناوی خشتە
ALTER TABLE table_name RENAME TO new_table_name;

-- ══════════════════════════════════════════
-- كۆدی ١ – create database
-- ══════════════════════════════════════════
-- دروستكردنی بنكەی زانیاری نوێ
create database emp;

-- ══════════════════════════════════════════
-- كۆدی ٢ – drop database
-- ══════════════════════════════════════════
-- سڕینەوەی بنكەی زانیاری بە تەواوی
drop database emp;

-- ══════════════════════════════════════════
-- كۆدی ٣ – select
-- ══════════════════════════════════════════
-- هەموو ستوونەكان هەڵبژێرە
select * from employees;
-- ستوونی دیاریكراو هەڵبژێرە
select id, name from employees;

-- ══════════════════════════════════════════
-- كۆدی ٤ – where
-- ══════════════════════════════════════════
-- تەنها ئەو ڕیزانە پیشان بدە كە مەرجەكە تێدایەتەوە
select * from employees
where name = 'ali';

-- ══════════════════════════════════════════
-- كۆدی ٥ – update
-- ══════════════════════════════════════════
-- گۆڕینی ستوونێك یان چەند ستوون لە خشتەیەكدا
update employees
set salary = 500, city = 'kalar'
where id = 1;

-- ══════════════════════════════════════════
-- كۆدی ٦ – delete
-- ══════════════════════════════════════════
-- سڕینەوەی ڕیزی دیاریكراو بە مەرج
delete from employees
where id = 3;

-- ══════════════════════════════════════════
-- كۆدی ٧ – insert into
-- ══════════════════════════════════════════
-- زیادكردنی ڕیزی نوێ بۆ خشتەیەك
insert into employees (id, name, salary)
values (1, 'hana', 300);

-- ══════════════════════════════════════════
-- كۆدی ٨ – create table & alter table
-- ══════════════════════════════════════════
-- دروستكردنی خشتەی نوێ
create table employees (
  id int,
  name varchar(50),
  salary int
);
-- زیادكردنی ستوونی نوێ
alter table employees add email varchar(100);
-- سڕینەوەی ستوون
alter table employees drop column email;
-- گۆڕینی ناوی ستوون
alter table employees rename column name to fullname;

-- ══════════════════════════════════════════
-- كۆدی ٩ – modify, add constraint & rename table
-- ══════════════════════════════════════════
-- گۆڕینی جۆری داتا یان سنووری ستوون
alter table employees modify name varchar(100) not null;
-- زیادكردنی سنووری نوێ
alter table employees add constraint pk_id primary key (id);
-- گۆڕینی ناوی خشتە
alter table employees rename to employees;
